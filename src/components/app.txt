def get_db_connection(is_bem=False):
    return pymysql.connect(
        host=os.environ.get("DB_HOST"),
        user=os.environ.get("BEM_DB_USER") if is_bem else os.environ.get("DB_USER"),
        password=os.environ.get("BEM_DB_PASSWORD") if is_bem else os.environ.get("DB_PASSWORD"),
        database=os.environ.get("BEM_DB_NAME") if is_bem else os.environ.get("DB_NAME"),
        port=int(os.environ.get("DB_PORT", 3306)),
        cursorclass=pymysql.cursors.DictCursor
    )

def query_promise(sql, params=None, is_bem=False):
    connection = get_db_connection(is_bem)
    try:
        with connection.cursor() as cursor:
            cursor.execute(sql, params or ())
            if sql.strip().upper().startswith("SELECT") or sql.strip().upper().startswith("SHOW"):
                return cursor.fetchall()
            connection.commit()
            return {"affectedRows": cursor.rowcount, "insertId": cursor.lastrowid}
    finally:
        connection.close()

def bem_query_promise(sql, params=None):
    return query_promise(sql, params, is_bem=True)

@contextmanager
def transaction_scope(is_bem=False):
    """Provides an atomic transactional scope for multi-query operations."""
    connection = get_db_connection(is_bem)
    try:
        yield connection
        connection.commit()
    except Exception as e:
        connection.rollback()
        raise e
    finally:
        connection.close()

def normalize_text(value):
    if value is None or str(value).strip() == "":
        return None
    return str(value).strip()

def normalize_number(value):
    if value is None or str(value).strip() == "":
        return None
    try:
        return float(value)
    except ValueError:
        return None

def parse_standard_date(value):
    if not value:
        return None
    try:
        date_str = str(value).strip()[:10]
        parsed_date = pd.to_datetime(date_str).strftime('%Y-%m-%d')
        return parsed_date
    except Exception:
        return str(value)

def get_current_user(req):
    return getattr(req, 'user', {}).get('username', 'system')

def authenticate_token(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        auth_header = request.headers.get("Authorization")
        if not auth_header or not auth_header.startswith("Bearer "):
            return jsonify({"message": "No token provided."}), 401
        
        token = auth_header.split(" ")[1]
        try:
            user = jwt.decode(token, JWT_SECRET, algorithms=["HS256"])
            request.user = user  
        except jwt.ExpiredSignatureError:
            return jsonify({"message": "Session expired. Please log in again."}), 403
        except jwt.InvalidTokenError:
            return jsonify({"message": "Invalid token."}), 403
        
        return f(*args, **kwargs)
    return decorated

def require_admin(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        if getattr(request, 'user', {}).get('access_right') != 'admin':
            return jsonify({"message": "Forbidden: Admins only."}), 403
        return f(*args, **kwargs)
    return decorated

@app.route('/api/login', methods=['POST'])
def login():
    data = request.get_json()
    username = data.get('username')
    password = data.get('password')

    if not username or not password:
        return jsonify({"success": False, "message": "Username and password are required."}), 400

    try:
        server = Server(LDAP_URL, get_info=ALL)
        
        if LDAP_BIND_DN and LDAP_BIND_PASSWORD:
            conn = Connection(server, user=LDAP_BIND_DN, password=LDAP_BIND_PASSWORD, auto_bind=True)
        else:
            conn = Connection(server, auto_bind=True)
            
        search_filter = f"(uid={username})"
        conn.search(BASE_DN, search_filter, attributes=['dn'])
        
        if not conn.entries:
            return jsonify({"success": False, "message": "User not found."}), 401
            
        user_dn = conn.entries[0].entry_dn
        
        user_conn = Connection(server, user=user_dn, password=password)
        if not user_conn.bind():
            return jsonify({"success": False, "message": "Invalid username or password."}), 401
            
        results = query_promise(f"SELECT access_right FROM {ACCESS_TABLE} WHERE username = %s", [username])
        access_right = results[0]['access_right'] if results else ''
        
        expiration = datetime.now(timezone.utc) + timedelta(hours=4)
        token = jwt.encode(
            {"username": username, "access_right": access_right, "exp": expiration},
            JWT_SECRET,
            algorithm="HS256"
        )
        
        return jsonify({
            "success": True, 
            "message": "LDAP authentication successful.", 
            "token": token, 
            "access_right": access_right
        })
        
    except Exception as e:
        print(f"LDAP Error: {str(e)}")
        return jsonify({"success": False, "message": "LDAP connection failed."}), 500

def get_table_columns(table_name, db_name=None):
    db_to_use = db_name or os.environ.get("DB_NAME")
    sql = """
        SELECT COLUMN_NAME
        FROM information_schema.COLUMNS
        WHERE TABLE_SCHEMA = %s AND TABLE_NAME = %s
        ORDER BY ORDINAL_POSITION
    """
    rows = query_promise(sql, [db_to_use, table_name])
    return [r['COLUMN_NAME'] for r in rows]

@app.route('/api/manager-yearly', methods=['GET'])
@authenticate_token
def get_manager_yearly():
    try:
        projects = bem_query_promise(f"SELECT * FROM {MANAGER_PROJECTS_TABLE} ORDER BY project_name ASC")
        yearly_values = bem_query_promise(f"SELECT * FROM {MANAGER_YEARLY_TABLE}")
        
        metrics = bem_query_promise(f"""
            SELECT m.* FROM {MANAGER_METRICS_TABLE} m
            INNER JOIN (
                SELECT project_id, MAX(reporting_year) as max_year 
                FROM {MANAGER_METRICS_TABLE} GROUP BY project_id
            ) latest ON m.project_id = latest.project_id AND m.reporting_year = latest.max_year
        """)

        comments = bem_query_promise(f"""
            SELECT c.* FROM {MANAGER_COMMENTS_TABLE} c
            INNER JOIN (
                SELECT project_id, MAX(created_at) as max_date 
                FROM {MANAGER_COMMENTS_TABLE} GROUP BY project_id
            ) latest ON c.project_id = latest.project_id AND c.created_at = latest.max_date
        """)

        phases = bem_query_promise(f"""
            SELECT project_id, phase_name, DATE_FORMAT(MIN(start_date), '%Y-%m-%d') as actual_start 
            FROM {MANAGER_PHASES_TABLE} 
            WHERE phase_name IN ('G1', 'G2', 'G3') 
            GROUP BY project_id, phase_name
        """)

        result = []
        for proj in projects:
            p_id = proj['project_id']
            proj_metrics = next((m for m in metrics if m['project_id'] == p_id), {})
            proj_comment = next((c for c in comments if c['project_id'] == p_id), {})
            
            g1 = next((p for p in phases if p['project_id'] == p_id and p['phase_name'] == 'G1'), None)
            g2 = next((p for p in phases if p['project_id'] == p_id and p['phase_name'] == 'G2'), None)
            g3 = next((p for p in phases if p['project_id'] == p_id and p['phase_name'] == 'G3'), None)

            tg1 = proj.get('target_g1_date')
            tg2 = proj.get('target_g2_date')
            tg3 = proj.get('target_g3_date')
            tclosed = proj.get('target_closed_date')

            payload = {
                "projectId": p_id,
                "projectName": proj.get('project_name'),
                "projectStatus": proj.get('project_status'),
                "pillars": proj.get('pillar'),
                "sites": proj.get('site'),
                "currentPmoGate": proj.get('current_pmo_gate'),
                "dtitInvolved": proj.get('dtit_involved'),
                "aiAaAType": proj.get('ai_type'),
                "foakNoak": proj.get('foak_noak'),
                
                "targetG1Date": tg1.strftime('%Y-%m-%d') if hasattr(tg1, 'strftime') else tg1,
                "targetG2Date": tg2.strftime('%Y-%m-%d') if hasattr(tg2, 'strftime') else tg2,
                "targetG3Date": tg3.strftime('%Y-%m-%d') if hasattr(tg3, 'strftime') else tg3,
                "targetClosedDate": tclosed.strftime('%Y-%m-%d') if hasattr(tclosed, 'strftime') else tclosed,
                
                "actualG1Date": g1['actual_start'] if g1 else None,
                "actualG2Date": g2['actual_start'] if g2 else None,
                "actualG3Date": g3['actual_start'] if g3 else None,

                "comment": proj_comment.get('comment_text'),

                "capacityGainValue": proj_metrics.get('capacity_gain_value'),
                "capacityGainPercent": proj_metrics.get('capacity_gain_pct'),
                "dlValue": proj_metrics.get('dl_value'),
                "dlEquivalent": proj_metrics.get('dl_equivalent'),
                "idlValue": proj_metrics.get('idl_value'),
                "idlFte": proj_metrics.get('idl_fte'),
                "yieldValue": proj_metrics.get('yield_value'),
                "yieldPercentGain": proj_metrics.get('yield_gain_pct'),
                "qualityValue": proj_metrics.get('quality_value'),
                "qualityCases": proj_metrics.get('quality_cases')
            }

            for y in [val for val in yearly_values if val['project_id'] == p_id]:

@app.route('/api/project-values', methods=['POST'])
@authenticate_token
def create_project_value():
    try:
        data = request.get_json()
        p_code = normalize_text(data.get('project_code'))
        p_name = normalize_text(data.get('project_name'))
        month_end = parse_standard_date(data.get('month_end'))

        if not p_code and not p_name:
            return jsonify({"message": "project_code or project_name is required."}), 400

        project = None
        if p_code:
            res = query_promise(f"SELECT project_code, project_name FROM {PROJECT_MASTER_TABLE} WHERE project_code = %s LIMIT 1", [p_code])
            project = res[0] if res else None
        elif p_name:
            res = query_promise(f"SELECT project_code, project_name FROM {PROJECT_MASTER_TABLE} WHERE project_name = %s LIMIT 1", [p_name])
            project = res[0] if res else None

        if not project:
            return jsonify({"message": "Selected project does not exist in project master."}), 400

        existing = query_promise(
            f"SELECT metric_id FROM {PROJECT_VALUES_TABLE} WHERE project_code = %s AND CAST(month_end AS DATE) = CAST(%s AS DATE)",
            [project['project_code'], month_end]
        )
        if existing:
            return jsonify({"message": f"Data for {project['project_name']} on month ending {month_end} already exists. Please select it from the table and click 'Update' instead."}), 409

        result = query_promise(f"""
            INSERT INTO {PROJECT_VALUES_TABLE}
            (project_code, month_end, capacity_improvement, people_productivity, quality_improvement, yield_savings)
            VALUES (%s, %s, %s, %s, %s, %s)
        """, [
            project['project_code'], month_end,
            normalize_number(data.get('capacity_improvement')),
            normalize_number(data.get('people_productivity')),
            normalize_number(data.get('quality_improvement')),
            normalize_number(data.get('yield_savings'))
        ])

        new_row = query_promise(f"""
            SELECT v.metric_id, v.project_code, m.project_name, m.project_type, m.project_description, m.project_status, m.use_case_investment,
                   DATE_FORMAT(v.month_end, '%Y-%m-%d') AS month_end, v.capacity_improvement, v.people_productivity, v.quality_improvement, v.yield_savings
            FROM {PROJECT_VALUES_TABLE} v
            LEFT JOIN {PROJECT_MASTER_TABLE} m ON v.project_code = m.project_code
            WHERE v.metric_id = %s
        """, [result['insertId']])

        return jsonify(new_row[0]), 201

    except Exception as e:
        return jsonify({"message": str(e)}), 500

@app.route('/api/project-values/import', methods=['POST'])
@authenticate_token
def import_project_values():
    if 'file' not in request.files:
        return jsonify({"message": "No file uploaded."}), 400

    file = request.files['file']
    overwrite = request.form.get('overwrite') == 'true'
    current_user = get_current_user(request)
    errors = []

    try:
        df = pd.read_excel(file)
        df = df.fillna('')
        
        first_row_columns = df.columns.tolist()
        minimum_columns = ['month_end', 'capacity_improvement', 'people_productivity', 'quality_improvement', 'yield_savings']
        missing_minimum = [col for col in minimum_columns if col not in first_row_columns]

        if missing_minimum:
            return jsonify({"message": f"Excel missing columns: {', '.join(missing_minimum)}"}), 400

        if 'project_code' not in first_row_columns and 'project_name' not in first_row_columns:
            return jsonify({"message": "Excel must contain either 'project_code' or 'project_name'."}), 400

        inserted = 0
        updated = 0
        failed = 0
        parsed_map = {}

        for index, row in df.iterrows():
            try:
                p_code = normalize_text(row.get('project_code'))
                p_name = normalize_text(row.get('project_name'))
                month_end = parse_standard_date(row.get('month_end'))

                project = None
                if p_code:
                    res = query_promise(f"SELECT project_code FROM {PROJECT_MASTER_TABLE} WHERE project_code = %s LIMIT 1", [p_code])
                    project = res[0] if res else None
                elif p_name:
                    res = query_promise(f"SELECT project_code FROM {PROJECT_MASTER_TABLE} WHERE project_name = %s LIMIT 1", [p_name])
                    project = res[0] if res else None

                if not project:
                    raise Exception('Project not found in project master.')

                resolved_code = project['project_code']
                unique_key = f"{resolved_code}_{month_end}"
                
                parsed_map[unique_key] = {
                    "resolved_project_code": resolved_code,
                    "month_end": month_end,
                    "capacity_improvement": normalize_number(row.get('capacity_improvement')),
                    "people_productivity": normalize_number(row.get('people_productivity')),
                    "quality_improvement": normalize_number(row.get('quality_improvement')),
                    "yield_savings": normalize_number(row.get('yield_savings'))
                }

            except Exception as row_err:
                failed += 1
                errors.append({"row": index + 2, "message": str(row_err)})

        unique_payloads = list(parsed_map.values())

        for payload in unique_payloads:
            existing = query_promise(
                f"SELECT metric_id FROM {PROJECT_VALUES_TABLE} WHERE project_code = %s AND CAST(month_end AS DATE) = CAST(%s AS DATE)",
                [payload['resolved_project_code'], payload['month_end']]
            )
            if existing:
                if not overwrite:
                    return jsonify({"requires_confirmation": True, "message": "Duplicate months detected. Waiting for user confirmation."}), 409
                payload['metric_id'] = existing[0]['metric_id']
                payload['isUpdate'] = True
            else:
                payload['isUpdate'] = False

        for payload in unique_payloads:
            if payload['isUpdate']:
                query_promise(f"""
                    INSERT INTO {HISTORY_TABLE} (
                        metric_id, project_code, month_end, capacity_improvement, yield_savings, people_productivity, quality_improvement, changed_by, action_type
                    )
                    SELECT metric_id, project_code, month_end, capacity_improvement, yield_savings, people_productivity, quality_improvement, %s, 'UPDATE'
                    FROM {PROJECT_VALUES_TABLE} WHERE metric_id = %s
                """, [current_user, payload['metric_id']])
                
                query_promise(f"""
                    UPDATE {PROJECT_VALUES_TABLE} SET capacity_improvement=%s, people_productivity=%s, quality_improvement=%s, yield_savings=%s WHERE metric_id=%s
                """, [payload['capacity_improvement'], payload['people_productivity'], payload['quality_improvement'], payload['yield_savings'], payload['metric_id']])
                updated += 1
            else:
                query_promise(f"""
                    INSERT INTO {PROJECT_VALUES_TABLE} (project_code, month_end, capacity_improvement, people_productivity, quality_improvement, yield_savings) 
                    VALUES (%s, %s, %s, %s, %s, %s)
                """, [payload['resolved_project_code'], payload['month_end'], payload['capacity_improvement'], payload['people_productivity'], payload['quality_improvement'], payload['yield_savings']])
                inserted += 1

        return jsonify({
            "message": f"Import completed. Inserted: {inserted}, Updated: {updated}, Failed: {failed}.",
            "summary": {"total": len(df), "inserted": inserted, "updated": updated, "failed": failed},
            "errors": errors
        })

    except Exception as e:
        print(f"Excel Error: {str(e)}")
        return jsonify({"message": str(e) or 'Excel import failed.'}), 500

def update_project_master(project_code):
    try:
        data = request.get_json()
        current_user = get_current_user(request)
        payload = {
            "project_code": normalize_text(data.get('project_code')),
            "project_name": normalize_text(data.get('project_name')),
            "project_description": normalize_text(data.get('project_description')),
            "project_type": normalize_text(data.get('project_type')),
            "use_case_investment": normalize_number(data.get('use_case_investment')),
            "project_status": normalize_text(data.get('project_status')) or 'Active'
        }

        validation_error = validate_project_master_payload(payload)
        if validation_error:
            return jsonify({"message": validation_error}), 400

        duplicate = query_promise(f"""
            SELECT project_code FROM {PROJECT_MASTER_TABLE} 
            WHERE (project_code = %s OR project_name = %s) AND project_code != %s LIMIT 1
        """, [payload['project_code'], payload['project_name'], project_code])
        
        if duplicate:
            return jsonify({"message": "Another project with the same code or name already exists."}), 409

        query_promise(f"""

                project_code, project_name, project_description, project_type, use_case_investment, project_status, changed_by, action_type

            UPDATE {PROJECT_MASTER_TABLE}
            SET project_code = %s, project_name = %s, project_description = %s, project_type = %s, use_case_investment = %s, project_status = %s

@app.route('/api/project-master/upload-excel', methods=['POST'])
@authenticate_token
@require_admin
def import_project_master():
    if 'file' not in request.files:
        return jsonify({"message": "No file uploaded."}), 400

    file = request.files['file']
    current_user = get_current_user(request)

    try:
        df = pd.read_excel(file)
        df = df.fillna('')
        
        first_row_columns = df.columns.tolist()
        expected_columns = ['project_code', 'project_name', 'project_type']
        missing = [col for col in expected_columns if col not in first_row_columns]

        if missing:
            return jsonify({"message": f"Excel missing columns: {', '.join(missing)}"}), 400

        inserted = 0
        updated = 0

        for _, row in df.iterrows():
            payload = {
                "project_code": normalize_text(row.get('project_code')),
                "project_name": normalize_text(row.get('project_name')),
                "project_description": normalize_text(row.get('project_description')),
                "project_type": normalize_text(row.get('project_type')),
                "use_case_investment": normalize_number(row.get('use_case_investment')),
                "project_status": normalize_text(row.get('project_status')) or 'Active'
            }

            if validate_project_master_payload(payload):
                continue

            existing = query_promise(
                f"SELECT project_code FROM {PROJECT_MASTER_TABLE} WHERE project_code = %s LIMIT 1",
                [payload['project_code']]
            )

            if existing:
                query_promise(f"""
                    INSERT INTO {MASTER_HISTORY_TABLE} (
                        project_code, project_name, project_description, project_type, use_case_investment, project_status, changed_by, action_type
                    )
                    SELECT project_code, project_name, project_description, project_type, use_case_investment, project_status, %s, 'UPDATE'
                    FROM {PROJECT_MASTER_TABLE} WHERE project_code = %s
                """, [current_user, payload['project_code']])

                query_promise(f"""
                    UPDATE {PROJECT_MASTER_TABLE}
                    SET project_name = %s, project_description = %s, project_type = %s, use_case_investment = %s, project_status = %s
                    WHERE project_code = %s
                """, [
                    payload['project_name'], payload['project_description'], payload['project_type'], 
                    payload['use_case_investment'] or 0, payload['project_status'], payload['project_code']
                ])
                updated += 1
            else:
                query_promise(f"""
                    INSERT INTO {PROJECT_MASTER_TABLE}
                    (project_code, project_name, project_description, project_type, use_case_investment, project_status)
                    VALUES (%s, %s, %s, %s, %s, %s)
                """, [
                    payload['project_code'], payload['project_name'], payload['project_description'], 
                    payload['project_type'], payload['use_case_investment'] or 0, payload['project_status']
                ])
                inserted += 1

        return jsonify({"message": f"Project master Excel upload completed. Inserted: {inserted}, Updated: {updated}."}), 200

    except Exception as e:
        print(f"Excel Error: {str(e)}")
        return jsonify({"message": str(e) or 'Excel upload failed.'}), 500

            "g3": float(r['g3_count']),

        } for r in rows]

@app.route('/api/manager-yearly/<project_id>', methods=['PUT'])
@authenticate_token  
@require_admin
def update_manager_yearly(project_id):
    try:
        data = request.get_json()
        current_user = get_current_user(request) 
        current_year = datetime.now().year
        
        target_g1_date = parse_standard_date(data.get('targetG1Date'))
        actual_g1_date = parse_standard_date(data.get('actualG1Date'))
        target_g2_date = parse_standard_date(data.get('targetG2Date'))
        actual_g2_date = parse_standard_date(data.get('actualG2Date'))
        target_g3_date = parse_standard_date(data.get('targetG3Date'))
        actual_g3_date = parse_standard_date(data.get('actualG3Date'))
        target_closed_date = parse_standard_date(data.get('targetClosedDate'))
        actual_closed_date = parse_standard_date(data.get('actualClosedDate'))

        with transaction_scope(is_bem=True) as conn:
            with conn.cursor() as cursor:
                # 0. Get Current State
                cursor.execute(f"""
                    SELECT 
                        p.project_name, p.project_status,
                        m.capacity_gain_value, m.capacity_gain_pct, m.dl_value, m.dl_equivalent, 
                        m.idl_value, m.idl_fte, m.yield_value, m.yield_gain_pct, 
                        m.quality_value, m.quality_cases, 
                        c.comment_text
                    FROM {MANAGER_PROJECTS_TABLE} p
                    LEFT JOIN {MANAGER_METRICS_TABLE} m ON p.project_id = m.project_id AND m.reporting_year = %s
                    LEFT JOIN {MANAGER_COMMENTS_TABLE} c ON p.project_id = c.project_id
                    WHERE p.project_id = %s
                    ORDER BY c.created_at DESC LIMIT 1
                """, [current_year, project_id])
                
                current_state = cursor.fetchall()

                if current_state:
                    curr = current_state[0]
                    cursor.execute(f"""
                        INSERT INTO {MANAGER_HISTORY_TABLE} (
                            project_id, project_name, project_status, capacity_gain_value, capacity_gain_pct, 
                            dl_value, dl_equivalent, idl_value, idl_fte, yield_value, yield_gain_pct, 
                            quality_value, quality_cases, comment_text, changed_by, action_type
                        ) VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, 'UPDATE')
                    """, [
                        project_id, curr.get('project_name'), curr.get('project_status'), 
                        curr.get('capacity_gain_value'), curr.get('capacity_gain_pct'),
                        curr.get('dl_value'), curr.get('dl_equivalent'), curr.get('idl_value'), curr.get('idl_fte'), 
                        curr.get('yield_value'), curr.get('yield_gain_pct'),
                        curr.get('quality_value'), curr.get('quality_cases'), curr.get('comment_text'), current_user
                    ])

                # 1. Update Project Master (Metadata)
                if data.get('projectName'):
                    new_status = normalize_text(data.get('projectStatus'))
                    cursor.execute(f"""
                        UPDATE {MANAGER_PROJECTS_TABLE} 
                        SET project_name = %s, project_status = %s, pillar = %s, site = %s, 
                            current_pmo_gate = %s, dtit_involved = %s, ai_type = %s, foak_noak = %s,
                            target_g1_date = %s, actual_g1_date = %s, 
                            target_g2_date = %s, actual_g2_date = %s, 
                            target_g3_date = %s, actual_g3_date = %s, 
                            target_closed_date = %s, actual_closed_date = %s
                        WHERE project_id = %s
                    """, [
                        normalize_text(data.get('projectName')), new_status, 
                        normalize_text(data.get('pillars')), normalize_text(data.get('sites')),
                        normalize_text(data.get('currentPmoGate')), normalize_text(data.get('dtitInvolved')), 
                        normalize_text(data.get('aiAaAType')), normalize_text(data.get('foakNoak')), 
                        target_g1_date, actual_g1_date, target_g2_date, actual_g2_date,
                        target_g3_date, actual_g3_date, target_closed_date, actual_closed_date,
                        project_id
                    ])

                    # Automate Actual Phase Tracking
                    old_status = current_state[0].get('project_status') if current_state else None
                    if old_status != new_status and new_status:
                        cursor.execute(f"UPDATE {MANAGER_PHASES_TABLE} SET end_date = CURDATE() WHERE project_id = %s AND end_date IS NULL", [project_id])
                        cursor.execute(f"INSERT INTO {MANAGER_PHASES_TABLE} (project_id, phase_name, start_date) VALUES (%s, %s, CURDATE())", [project_id, new_status])

                # 2. Update Yearly Savings safely
                year_keys = [k for k in data.keys() if k.startswith('year') and len(k) == 8]
                for key in year_keys:
                    year = int(key.replace('year', ''))
                    val = normalize_number(data.get(key))
                    if val is not None:
                        cursor.execute(f"SELECT project_id FROM {MANAGER_YEARLY_TABLE} WHERE project_id = %s AND savings_year = %s LIMIT 1", [project_id, year])
                        if cursor.fetchall():
                            cursor.execute(f"UPDATE {MANAGER_YEARLY_TABLE} SET savings_value = %s WHERE project_id = %s AND savings_year = %s", [val, project_id, year])
                        else:
                            cursor.execute(f"INSERT INTO {MANAGER_YEARLY_TABLE} (project_id, savings_year, savings_value) VALUES (%s, %s, %s)", [project_id, year, val])

                # 3. Update Metrics (KPIs)
                cursor.execute(f"SELECT metrics_id FROM {MANAGER_METRICS_TABLE} WHERE project_id = %s AND reporting_year = %s LIMIT 1", [project_id, current_year])
                existing_metric = cursor.fetchall()

                metrics_params = [
                    normalize_number(data.get('capacityGainValue')), normalize_number(data.get('capacityGainPercent')),
                    normalize_number(data.get('dlValue')), normalize_number(data.get('dlEquivalent')),
                    normalize_number(data.get('idlValue')), normalize_number(data.get('idlFte')),
                    normalize_number(data.get('yieldValue')), normalize_number(data.get('yieldPercentGain')),
                    normalize_number(data.get('qualityValue')), normalize_number(data.get('qualityCases'))
                ]
                
                if existing_metric:
                    metrics_params.append(existing_metric[0]['metrics_id'])
                    cursor.execute(f"""
                        UPDATE {MANAGER_METRICS_TABLE} SET 
                        capacity_gain_value=%s, capacity_gain_pct=%s, dl_value=%s, dl_equivalent=%s, 
                        idl_value=%s, idl_fte=%s, yield_value=%s, yield_gain_pct=%s, quality_value=%s, quality_cases=%s 
                        WHERE metrics_id=%s
                    """, metrics_params)
                else:
                    metrics_params = [project_id, current_year] + metrics_params
                    cursor.execute(f"""
                        INSERT INTO {MANAGER_METRICS_TABLE} 
                        (project_id, reporting_year, capacity_gain_value, capacity_gain_pct, dl_value, dl_equivalent, idl_value, idl_fte, yield_value, yield_gain_pct, quality_value, quality_cases)
                        VALUES (%s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s, %s)
                    """, metrics_params)

                # 4. Update Comments
                comment_text = normalize_text(data.get('comment'))
                if comment_text is not None:
                    cursor.execute(f"SELECT comment_id FROM {MANAGER_COMMENTS_TABLE} WHERE project_id = %s ORDER BY created_at DESC LIMIT 1", [project_id])
                    existing_comment = cursor.fetchall()
                    if existing_comment:
                        cursor.execute(f"UPDATE {MANAGER_COMMENTS_TABLE} SET comment_text = %s WHERE comment_id = %s", [comment_text, existing_comment[0]['comment_id']])
                    else:
                        cursor.execute(f"INSERT INTO {MANAGER_COMMENTS_TABLE} (project_id, comment_text) VALUES (%s, %s)", [project_id, comment_text])

        return jsonify({"message": "Project updated successfully."}), 200

    except Exception as e:
        print(f"Error updating project: {str(e)}")
        return jsonify({"message": str(e)}), 500
